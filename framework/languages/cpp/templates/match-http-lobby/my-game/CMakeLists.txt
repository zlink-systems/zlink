cmake_minimum_required(VERSION 3.24)
project(MyGame LANGUAGES CXX)
# bootstrap.cmake installs the pinned framework prebuilt, which also carries
# Boost and nlohmann_json; CMakePresets.json puts its prefix on the search path.
find_package(zlink_framework CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(MyGame.Shared INTERFACE)
target_include_directories(MyGame.Shared INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(MyGame.Shared INTERFACE cxx_std_20)
target_link_libraries(MyGame.Shared INTERFACE Boost::headers nlohmann_json::nlohmann_json Threads::Threads)
foreach(role session match lobby)
  add_executable(my-game-${role} server/${role}/main.cpp)
  target_link_libraries(my-game-${role} PRIVATE MyGame.Shared zlink::framework zlink::framework_locations_redis)
endforeach()
add_executable(my-game-e2e e2e/main.cpp)
target_link_libraries(my-game-e2e PRIVATE MyGame.Shared zlink::stream_connector)
set(game_targets my-game-session my-game-match my-game-lobby my-game-e2e)

include(CTest)
if(BUILD_TESTING)
  add_executable(my-game-rules e2e/domain_tests.cpp)
  target_link_libraries(my-game-rules PRIVATE MyGame.Shared zlink::framework zlink::framework_locations_redis)
  add_test(NAME room-rules COMMAND my-game-rules)
  add_executable(my-game-wire e2e/wire_test.cpp)
  target_link_libraries(my-game-wire PRIVATE MyGame.Shared zlink::framework)
  add_test(NAME wire-golden COMMAND my-game-wire)
  list(APPEND game_targets my-game-rules my-game-wire)
endif()
foreach(target IN LISTS game_targets)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /bigobj)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(WIN32)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:${target}> $<TARGET_FILE_DIR:${target}>
      COMMAND_EXPAND_LISTS)
  endif()
endforeach()

if(WIN32)
  target_compile_definitions(MyGame.Shared INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(MyGame.Shared INTERFACE ws2_32 mswsock)
endif()
