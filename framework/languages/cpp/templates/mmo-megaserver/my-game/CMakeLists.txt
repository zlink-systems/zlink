cmake_minimum_required(VERSION 3.24)
if(DEFINED ENV{VCPKG_OVERLAY_PORTS})
  file(TO_CMAKE_PATH "$ENV{VCPKG_OVERLAY_PORTS}" VCPKG_OVERLAY_PORTS)
endif()
project(MyGame LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vcpkg.json" manifest)
string(JSON framework_version GET "${manifest}" overrides 0 version)
find_package(zlink_framework ${framework_version} EXACT CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(MyGame.Shared INTERFACE)
target_include_directories(MyGame.Shared INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(MyGame.Shared INTERFACE cxx_std_20)
target_link_libraries(MyGame.Shared INTERFACE Boost::headers nlohmann_json::nlohmann_json Threads::Threads)
foreach(role session region instance)
  add_executable(my-game-${role} server/${role}/main.cpp)
  target_link_libraries(my-game-${role} PRIVATE MyGame.Shared zlink::framework zlink::framework_locations_redis)
endforeach()
add_executable(my-game-e2e e2e/main.cpp)
target_link_libraries(my-game-e2e PRIVATE MyGame.Shared zlink::stream_connector)
foreach(target my-game-session my-game-region my-game-instance my-game-e2e)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /bigobj)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(WIN32)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:${target}> $<TARGET_FILE_DIR:${target}>
      COMMAND_EXPAND_LISTS)
  endif()
endforeach()
