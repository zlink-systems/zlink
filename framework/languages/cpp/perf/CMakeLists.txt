cmake_minimum_required(VERSION 3.24)
project(zlink_framework_perf LANGUAGES C CXX)
set(ZLINK_FRAMEWORK_CPP_BUILD_TESTS OFF CACHE BOOL "Perf dependency only")
set(ZLINK_FRAMEWORK_CPP_BUILD_FOUNDATION_TESTS OFF CACHE BOOL "Perf dependency only" FORCE)
set(ZLINK_FRAMEWORK_CPP_BUILD_SAMPLES OFF CACHE BOOL "Perf dependency only" FORCE)
set(ZLINK_FRAMEWORK_CPP_BUILD_E2E OFF CACHE BOOL "Perf dependency only" FORCE)
set(ZLINK_FRAMEWORK_CPP_BUILD_CROSS_LANGUAGE OFF CACHE BOOL "Perf dependency only" FORCE)
set(ZLINK_FRAMEWORK_CPP_USE_BINDINGS_SOURCE OFF CACHE BOOL "Use official binding package" FORCE)
set(ZLINK_STREAM_CONNECTOR_BUILD_E2E_CLIENT OFF CACHE BOOL "Public connector only")
set(ZLINK_STREAM_CONNECTOR_BUILD_UNREAL OFF CACHE BOOL "Perf dependency only" FORCE)
set(ZLINK_STREAM_CONNECTOR_BUILD_GODOT OFF CACHE BOOL "Perf dependency only" FORCE)
set(ZLINK_STREAM_CONNECTOR_BUILD_AXMOL OFF CACHE BOOL "Perf dependency only" FORCE)
# A complete packaged Boost tree is shared by all targets, including multiprecision.
set(ZLINK_FRAMEWORK_CPP_USE_SYSTEM_BOOST ON CACHE BOOL "Perf uses complete packaged Boost headers" FORCE)
find_package(Boost REQUIRED)
add_subdirectory(".." "${CMAKE_CURRENT_BINARY_DIR}/framework" EXCLUDE_FROM_ALL)
find_package(opentelemetry-cpp CONFIG REQUIRED COMPONENTS sdk)
add_executable(zlink_framework_perf main.cpp)
target_compile_features(zlink_framework_perf PRIVATE cxx_std_20)
target_link_libraries(zlink_framework_perf PRIVATE zlink::framework zlink::framework_locations_redis zlink::stream_connector opentelemetry-cpp::metrics)
target_compile_definitions(zlink_framework_perf PRIVATE ZLINK_PERF_CONTRACT_DIR="${CMAKE_CURRENT_LIST_DIR}/../../../perf-contract")
set_target_properties(zlink_framework_perf PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
if(COMMAND zlink_framework_cpp_add_boost_headers)
  zlink_framework_cpp_add_boost_headers(zlink_framework_perf)
endif()
