    std::string_view input;
    std::size_t position = 0;
    std::string scratch;

    unsigned char peek () const
    {
        return position == input.size () ? 0 : static_cast<unsigned char> (input[position]);
    }
    void whitespace ()
    {
        while (position < input.size ()) {
            const auto c = input[position];
            if (c != ' ' && c != '\t' && c != '\r' && c != '\n')
                break;
            ++position;
        }
    }
    bool consume (char c)
    {
        whitespace ();
        if (position == input.size () || input[position] != c)
            return reject ("invalid JSON");
        ++position;
        return true;
    }
    bool hex_quad (std::uint32_t &value)
    {
        value = 0;
        for (int i = 0; i != 4; ++i) {
            if (position == input.size ())
                return false;
            const auto c = input[position++];
            unsigned digit;
            if (c >= '0' && c <= '9')
                digit = c - '0';
            else if (c >= 'a' && c <= 'f')
                digit = c - 'a' + 10;
            else if (c >= 'A' && c <= 'F')
                digit = c - 'A' + 10;
            else
                return false;
            value = (value << 4) | digit;
        }
        return true;
    }
    bool unicode_escape ()
    {
        std::uint32_t code;
        if (!hex_quad (code))
            return false;
        if (code >= 0xd800 && code <= 0xdbff) {
            if (input.size () - position < 2 || input[position] != '\\'
                || input[position + 1] != 'u')
                return false;
            position += 2;
            std::uint32_t low;
            if (!hex_quad (low) || low < 0xdc00 || low > 0xdfff)
                return false;
            code = 0x10000 + ((code - 0xd800) << 10) + low - 0xdc00;
        } else if (code >= 0xdc00 && code <= 0xdfff)
            return false;
        if (code <= 0x7f)
            scratch += static_cast<char> (code);
        else {
            if (code <= 0x7ff)
                scratch += static_cast<char> (0xc0 | (code >> 6));
            else {
                if (code <= 0xffff)
                    scratch += static_cast<char> (0xe0 | (code >> 12));
                else {
                    scratch += static_cast<char> (0xf0 | (code >> 18));
                    scratch += static_cast<char> (0x80 | ((code >> 12) & 0x3f));
                }
                scratch += static_cast<char> (0x80 | ((code >> 6) & 0x3f));
            }
            scratch += static_cast<char> (0x80 | (code & 0x3f));
        }
        return true;
    }
    bool string_token (std::string_view &text)
    {
        if (!consume ('"'))
            return false;
        const auto start = position;
        auto segment = position;
        bool escaped = false;
        scratch.clear ();
        while (position < input.size ()) {
            const auto c = static_cast<unsigned char> (input[position++]);
            if (c == '"') {
                const auto tail = input.substr (segment, position - segment - 1);
                if (escaped) {
                    scratch.append (tail);
                    text = scratch;
                } else
                    text = tail;
                return valid_utf8 (text) || reject ("invalid UTF-8");
            }
            if (c < 0x20)
                return reject ("invalid JSON string");
            if (c != '\\')
                continue;
            if (!escaped) {
                // Unescaping never increases the raw token's UTF-8 size.
                auto end = position;
                bool slash = true;
                for (; end < input.size (); ++end) {
                    const auto next = input[end];
                    if (next == '"' && !slash)
                        break;
                    slash = next == '\\' && !slash;
                }
                scratch.reserve (end - start);
                escaped = true;
            }
            scratch.append (input.substr (segment, position - segment - 1));
            if (position == input.size ())
                return reject ("invalid JSON escape");
            switch (input[position++]) {
                case '"': scratch += '"'; break;
                case '\\': scratch += '\\'; break;
                case '/': scratch += '/'; break;
                case 'b': scratch += '\b'; break;
                case 'f': scratch += '\f'; break;
                case 'n': scratch += '\n'; break;
                case 'r': scratch += '\r'; break;
                case 't': scratch += '\t'; break;
                case 'u':
                    if (!unicode_escape ())
                        return reject ("invalid JSON unicode escape");
                    break;
                default: return reject ("invalid JSON escape");
            }
            segment = position;
        }
        return reject ("unterminated JSON string");
    }
    static bool digit (unsigned char c) { return c >= '0' && c <= '9'; }
    bool number_token (sax_event &event)
    {
        const auto start = position;
        const bool negative = peek () == '-';
        if (negative)
            ++position;
        if (!digit (peek ()))
            return reject ("invalid JSON number");
        if (peek () == '0')
            ++position;
        else
            while (digit (peek ()))
                ++position;
        bool floating = false;
        if (peek () == '.') {
            floating = true;
            ++position;
            if (!digit (peek ()))
                return reject ("invalid JSON fraction");
            while (digit (peek ()))
                ++position;
        }
        if (peek () == 'e' || peek () == 'E') {
            floating = true;
            ++position;
            if (peek () == '+' || peek () == '-')
                ++position;
            if (!digit (peek ()))
                return reject ("invalid JSON exponent");
            while (digit (peek ()))
                ++position;
        }
        const auto text = input.substr (start, position - start);
        const auto end = text.data () + text.size ();
        if (!floating) {
            if (negative) {
                event.kind = sax_kind::integer;
                if (std::from_chars (text.data (), end, event.integer).ec == std::errc{})
                    return true;
            } else {
                event.kind = sax_kind::unsigned_integer;
                if (std::from_chars (text.data (), end, event.unsigned_integer).ec == std::errc{})
                    return true;
            }
        }
        // Preserve strtod rounding, overflow promotion and signed underflow.
        scratch.assign (text);
        const auto decimal = std::localeconv ()->decimal_point[0];
        if (decimal != '.')
            std::replace (scratch.begin (), scratch.end (), '.', decimal);
        event.kind = sax_kind::floating;
        event.floating = std::strtod (scratch.c_str (), nullptr);
        return valid_float (event.floating) || reject ("non-finite number");
    }
    bool read_value (destination next)
    {
        whitespace ();
        sax_event event{sax_kind::null};
        switch (peek ()) {
            case '{': event.kind = sax_kind::object; ++position; break;
            case '[': event.kind = sax_kind::array; ++position; break;
            case '"':
                event.kind = sax_kind::string;
                if (!string_token (event.text))
                    return false;
                break;
            case 'n':
            case 't':
            case 'f': {
                const std::string_view word = peek () == 'n' ? "null" : peek () == 't' ? "true" : "false";
                if (input.substr (position, word.size ()) != word)
                    return reject ("invalid JSON literal");
                position += word.size ();
                if (word != "null") {
                    event.kind = sax_kind::boolean;
                    event.boolean = word == "true";
                }
                break;
            }
            default:
                if (!number_token (event))
                    return false;
        }
        return next.accept (*this, next.value, event);
    }

  public:
    template <typename T>
    typed_reader (T &out, std::string_view bytes, std::string *failure) :
        root (target (out)), error (failure), input (bytes)
    {
        frames.reserve (initial_frame_capacity);
    }
    bool read ()
    {
        if (input.starts_with ("\xef\xbb\xbf"))
            return reject ("framework-json-v1 rejects a UTF-8 BOM");
        if (!read_value (root))
            return false;
        while (!frames.empty ()) {
            auto &parent = frames.back ();
            whitespace ();
            const auto close = parent.kind == sax_kind::object ? '}' : ']';
            if (peek () == close) {
                ++position;
                if (!parent.finish (*this, parent))
                    return false;
                frames.pop_back ();
                continue;
            }
            if (parent.after_value && !consume (','))
                return false;
            if (parent.kind == sax_kind::object) {
                std::string_view key;
                if (!string_token (key))
                    return false;
                if (!parent.key (parent, key))
                    return reject ("framework-json-v1 rejects duplicate properties");
                if (!consume (':'))
                    return false;
            }
            parent.after_value = true;
            const auto next = parent.child (parent);
            if (!read_value (next))
                return false;
        }
        whitespace ();
        // Preserve the previous input adapter's zero byte terminator behavior.
        return peek () == 0 || reject ("trailing JSON value");
    }
